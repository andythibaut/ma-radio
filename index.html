function doPlay(st, url, myToken, urlIndex){ if(myToken!==token) return; if(hls){ hls.destroy(); hls=null; } if(audio){ audio.pause(); audio.src=""; audio=null; } const vol = $("volSlider").value/100; function tryNext(){ if(myToken!==token) return; const next = urlIndex + 1; if(!st.isHLS && next < st.streams.length){ doPlay(st, st.streams[next], myToken, next); } else { showError(); } } if(st.isHLS){ function startHLS(){ if(myToken!==token) return; audio = new Audio(); audio.volume = vol; if(window.Hls && Hls.isSupported()){ hls = new Hls({enableWorker:false}); hls.loadSource(url); hls.attachMedia(audio); hls.on(Hls.Events.MANIFEST_PARSED, ()=>{ if(myToken!==token) return; audio.play() .then(()=>{ if(myToken!==token){audio.pause();return;} playing=true; setIcon(true); setStatus("playing"); }) .catch(()=>{ if(myToken===token) showError(); }); }); hls.on(Hls.Events.ERROR, (_,d)=>{ if(d.fatal && myToken===token) showError(); }); } else if(audio.canPlayType("application/vnd.apple.mpegurl")){ audio.src=url; audio.onplaying=()=>{ if(myToken!==token)return; playing=true; setIcon(true); setStatus("playing"); }; audio.onerror=()=>{ if(myToken===token) showError(); }; audio.play().catch(()=>{ if(myToken===token) showError(); }); } else { showError(); } } if(window.Hls){ startHLS(); } else{ const s=document.createElement("script"); s.src="https://cdn.jsdelivr.net/npm/hls.js@latest/dist/hls.min.js"; s.onload=startHLS; s.onerror=()=>{ if(myToken===token) showError(); }; document.head.appendChild(s); } return; } audio = new Audio(); audio.volume = vol; audio.src = url; audio.onplaying = ()=>{ if(myToken!==token)return; playing=true; setIcon(true); setStatus("playing"); }; audio.onerror = ()=>{ if(myToken===token) tryNext(); }; audio.onwaiting = ()=>{ if(playing && myToken===token) setStatus("loading"); }; audio.play().catch(()=>{ if(myToken===token) tryNext(); }); }
